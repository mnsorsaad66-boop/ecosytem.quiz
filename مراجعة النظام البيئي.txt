        // Web Audio API for Simple Sound Effects
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSound(type) {
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);

            if (type === 'correct') {
                // Happy Arpeggio
                osc.type = 'sine';
                osc.frequency.setValueAtTime(523.25, audioCtx.currentTime); // C5
                osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.1); // E5
                osc.frequency.setValueAtTime(783.99, audioCtx.currentTime + 0.2); // G5
                osc.frequency.setValueAtTime(1046.50, audioCtx.currentTime + 0.3); // C6
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.5);
                osc.start(audioCtx.currentTime);
                osc.stop(audioCtx.currentTime + 0.5);
            } else if (type === 'wrong') {
                // Low Buzz
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(180, audioCtx.currentTime);
                osc.frequency.setValueAtTime(140, audioCtx.currentTime + 0.15);
                gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
                osc.start(audioCtx.currentTime);
                osc.stop(audioCtx.currentTime + 0.4);
            } else if (type === 'win') {
                // Victory Fanfare
                const notes = [440, 554.37, 659.25, 880];
                notes.forEach((freq, idx) => {
                    const o = audioCtx.createOscillator();
                    const g = audioCtx.createGain();
                    o.connect(g);
                    g.connect(audioCtx.destination);
                    o.frequency.value = freq;
                    g.gain.setValueAtTime(0.2, audioCtx.currentTime + idx * 0.12);
                    g.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + idx * 0.12 + 0.3);
                    o.start(audioCtx.currentTime + idx * 0.12);
                    o.stop(audioCtx.currentTime + idx * 0.12 + 0.3);
                });
            }
        }

                explanation: "التلوث يضر بالأنظمة البيئية والمخلوقات الحية المعتمدة عليها."
            }
        ];

        let currentQuestionIndex = 0;
        let score = 0;
        let studentName = "";
        let selectedOption = null;

        // DOM Elements
        const startScreen = document.getElementById('start-screen');
        const quizScreen = document.getElementById('quiz-screen');
        const certificateScreen = document.getElementById('certificate-screen');
        const studentNameInput = document.getElementById('student-name-input');
        const nameError = document.getElementById('name-error');
        
        const displayStudentName = document.getElementById('display-student-name');
        const currentQuestionNumEl = document.getElementById('current-question-num');
        const totalQuestionsNumEl = document.getElementById('total-questions-num');
        const scoreDisplay = document.getElementById('score-display');
        const progressBar = document.getElementById('progress-bar');
        
        const questionCategory = document.getElementById('question-category');
        const questionText = document.getElementById('question-text');
        const optionsContainer = document.getElementById('options-container');
        const feedbackBox = document.getElementById('feedback-box');
        const feedbackTitle = document.getElementById('feedback-title');
        const feedbackDetail = document.getElementById('feedback-detail');
        const feedbackIcon = document.getElementById('feedback-icon');
        const nextBtn = document.getElementById('next-btn');

        // Start Quiz Logic
        function startQuiz() {
            const name = studentNameInput.value.trim();
            if (!name) {
                nameError.classList.remove('hidden');
                studentNameInput.focus();
                return;
            }
            
            nameError.classList.add('hidden');
            studentName = name;
            displayStudentName.textContent = studentName;

            currentQuestionIndex = 0;
            score = 0;
            scoreDisplay.textContent = score;
            totalQuestionsNumEl.textContent = quizQuestions.length;

            startScreen.classList.add('hidden');
            quizScreen.classList.remove('hidden');

            loadQuestion();
        }

        // Load Single Question
        function loadQuestion() {
            const q = quizQuestions[currentQuestionIndex];
            selectedOption = null;

            // Reset UI
            feedbackBox.classList.add('hidden');
            nextBtn.classList.add('hidden');
            nextBtn.classList.remove('flex');
            
            // Progress Update
            currentQuestionNumEl.textContent = currentQuestionIndex + 1;
            const progressPercent = ((currentQuestionIndex) / quizQuestions.length) * 100;
            progressBar.style.width = `${progressPercent}%`;

            // Question Details
            questionCategory.textContent = q.category;
            questionText.textContent = q.question;

            // Options Build
            optionsContainer.innerHTML = '';
            q.options.forEach((opt, idx) => {
                const btn = document.createElement('button');
                btn.className = `btn-bounce w-full text-right p-4 rounded-2xl border-2 border-emerald-100 bg-white hover:bg-emerald-50 hover:border-emerald-300 font-bold text-gray-800 transition flex items-center justify-between shadow-sm group`;
                btn.onclick = () => selectOption(idx);
                btn.innerHTML = `
                    <span class="text-base sm:text-lg">${opt}</span>
                    <span class="w-7 h-7 rounded-full bg-emerald-100 text-emerald-700 flex items-center justify-center text-xs font-black group-hover:bg-emerald-600 group-hover:text-white transition">
                        ${String.fromCharCode(65 + idx)}
                    </span>
                `;
                optionsContainer.appendChild(btn);
            });
        }

        // Select Option Logic
        function selectOption(index) {
            if (selectedOption !== null) return; // Prevent multiple clicks
            selectedOption = index;

            const q = quizQuestions[currentQuestionIndex];
            const buttons = optionsContainer.children;

            // Disable all buttons
            for (let b of buttons) {
                b.disabled = true;
                b.classList.remove('btn-bounce');
            }

            if (index === q.correct) {
                // Correct Answer
                score += 10;
                scoreDisplay.textContent = score;
                playSound('correct');

                buttons[index].classList.remove('bg-white', 'border-emerald-100');
                buttons[index].classList.add('bg-emerald-500', 'text-white', 'border-emerald-600', 'correct-flash');
                buttons[index].querySelector('span:last-child').className = 'w-7 h-7 rounded-full bg-white text-emerald-700 flex items-center justify-center text-xs font-black';

                // Confetti FX
                triggerConfetti();

                // Show Positive Feedback
                feedbackBox.className = "mt-4 p-4 rounded-2xl transition-all duration-300 flex items-start gap-3 bg-emerald-100 text-emerald-900 border border-emerald-300";
                feedbackIcon.innerHTML = `<i class="fa-solid fa-circle-check text-emerald-600"></i>`;
                feedbackTitle.textContent = "إجابة صحيحة ومتميزة! 🌟";
                feedbackDetail.textContent = q.explanation;
                feedbackBox.classList.remove('hidden');

            } else {
                // Wrong Answer
                playSound('wrong');

                buttons[index].classList.remove('bg-white', 'border-emerald-100');
                buttons[index].classList.add('bg-rose-500', 'text-white', 'border-rose-600');

                // Highlight Correct Option
                buttons[q.correct].classList.remove('bg-white', 'border-emerald-100');
                buttons[q.correct].classList.add('bg-emerald-100', 'text-emerald-900', 'border-emerald-400');

                // Show Encouraging Feedback
                feedbackBox.className = "mt-4 p-4 rounded-2xl transition-all duration-300 flex items-start gap-3 bg-amber-50 text-amber-900 border border-amber-200";
                feedbackIcon.innerHTML = `<i class="fa-solid fa-lightbulb text-amber-500"></i>`;
                feedbackTitle.textContent = "محاولة جيدة! الإجابة الصحيحة موضحة باللون الأخضر.";
                feedbackDetail.textContent = q.explanation;
                feedbackBox.classList.remove('hidden');
            }

            // Show Next Button
            nextBtn.classList.remove('hidden');
            nextBtn.classList.add('flex');
        }

        // Trigger Confetti Effect
        function triggerConfetti() {
            confetti({
                particleCount: 60,
                spread: 70,
                origin: { y: 0.7 },
                colors: ['#10B981', '#F59E0B', '#0284C7', '#34D399']
            });
        }

        // Next Question or Finish Quiz
        function nextQuestion() {
            currentQuestionIndex++;
            if (currentQuestionIndex < quizQuestions.length) {
                loadQuestion();
            } else {
                showCertificate();
            }
        }

        // Show Certificate Screen
        function showCertificate() {
            quizScreen.classList.add('hidden');
            certificateScreen.classList.remove('hidden');

            // Fill Certificate Data
            document.getElementById('cert-student-name').textContent = studentName;
            document.getElementById('cert-score').textContent = `${score} / ${quizQuestions.length * 10}`;

            const percentage = (score / (quizQuestions.length * 10)) * 100;
            const gradeEl = document.getElementById('cert-grade');

            if (percentage >= 90) {
                gradeEl.textContent = "ممتاز مرتفع 🌟🏆";
            } else if (percentage >= 75) {
                gradeEl.textContent = "جيد جداً مرتفع ⭐";
            } else {
                gradeEl.textContent = "بطلة مستقبليّة 💪";
            }

            // Victory FX
            playSound('win');
            confetti({
                particleCount: 150,
                spread: 100,
                origin: { y: 0.5 }
            });
        }

        // Print / Save PDF
        function printCertificate() {
            window.print();
        }

        // Restart Game
        function restartQuiz() {
            certificateScreen.classList.add('hidden');
            startScreen.classList.remove('hidden');
            studentNameInput.value = '';
        }
    </script>
</body>
